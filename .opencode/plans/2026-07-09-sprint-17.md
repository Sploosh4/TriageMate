## Architecture Overview

### New 7th Scoring Dimension: "Security Context" (max 5 points)

The dimension combines telemetry + network signals into a single 0-5 score:

| Signal | Source | Max Contribution |
|---|---|---|
| Active Sentinel incident (high severity) | `synthetic_telemetry.py` | +3 |
| Vulnerable code IS loaded/running (EDR evidence) | `synthetic_telemetry.py` | +2 |
| High blast radius (many exposed peers) | `network_context.py` | +1 |

**Total weight:** 107 → 112 (clamped to 0-100 as before)

### Data Flow

```
ingest_vulnerability()
  │
  ├─ Pre-Tier (mission tags)
  ├─ Asset enrichment (CMDB, data_classification, internet_facing)
  │
  ├─ **NEW: get_telemetry_context(cve_id, hostname, cvss, kev, epss, description)**
  │     → telemetry: { sentinel_incidents, edr_alerts, process_evidence, 
  │                    threat_score (0.0-1.0) }
  │
  ├─ **NEW: get_network_context(hostname, asset_ctx)**
  │     → network: { zone, peer_count, exposed_peers, blast_radius_score (0.0-1.0) }
  │
  ├─ Tier 0 (deterministic rules — can use telemetry signals)
  ├─ Tier 1 (XGBoost — telemetry feeds into exploitation probability)
  ├─ Tier 2 (LLM — can reference telemetry in conflict assessment)
  │
  └─ calculate_final_score()  ← NEW 7th dimension: security_context
       → security_context = round(threat_score * 3 + blast_radius_score * 2)  [max 5]
       → total dimensions: 7
       → weight sum: 112 (was 107)
       → clamped to 0-100
```

**Generation logic (probabilistic):**

```python
import hashlib, random

def _seed(cve_id: str, hostname: str) -> random.Random:
    """Deterministic seed from CVE+hostname for reproducibility in demos,
    but varies across different CVEs/hosts."""
    h = hashlib.sha256(f"{cve_id}:{hostname}:telemetry".encode()).digest()
    return random.Random(int.from_bytes(h[:8], 'big'))

def get_telemetry_context(cve_id, hostname, cvss_score, cisa_kev, epss_score, description):
    rng = _seed(cve_id, hostname)
    
    # Base probabilities from vulnerability characteristics
    p_sentinel_active = 0.0
    if cisa_kev:                      p_sentinel_active = 0.85
    elif epss_score > 0.5:            p_sentinel_active = 0.50
    elif epss_score > 0.1:            p_sentinel_active = 0.20
    elif cvss_score >= 9.0:           p_sentinel_active = 0.35
    
    # EDR evidence: is the vulnerable code actually running?
    has_rce = any(kw in description.lower() for kw in ['remote code execution', 'rce', 'arbitrary code', 'code execution'])
    has_service = any(kw in description.lower() for kw in ['service', 'daemon', 'server', 'web', 'portal', 'gateway'])
    has_library = any(kw in description.lower() for kw in ['library', 'module', 'component', 'dependency', 'package'])
    
    # Process evidence: more likely if it's a service/RCE CVE
    if has_rce and has_service:
        p_process_running = 0.70
        p_process_loaded = 0.85
    elif has_service:
        p_process_running = 0.60
        p_process_loaded = 0.70
    elif has_library:
        p_process_running = 0.30
        p_process_loaded = 0.50
    else:
        p_process_running = 0.15
        p_process_loaded = 0.25
    
    is_running = rng.random() < p_process_running
    is_loaded = rng.random() < p_process_loaded
    
    # Generate incidents
    incidents = []
    if rng.random() < p_sentinel_active:
        severity = "High" if cvss_score >= 9.0 else ("Medium" if cvss_score >= 7.0 else "Low")
        incidents = [_generate_synthetic_incident(rng, cve_id, hostname, severity)]
        # 30% chance of a second related incident
        if rng.random() < 0.30:
            incidents.append(_generate_synthetic_incident(rng, cve_id, hostname, "Medium"))
    
    # Generate EDR alerts
    alerts = []
    if is_running:
        alerts.append(_generate_edr_process_alert(rng, cve_id, hostname))
        if rng.random() < 0.40:
            alerts.append(_generate_edr_network_alert(rng, cve_id, hostname))
        if rng.random() < 0.25:
            alerts.append(_generate_edr_module_alert(rng, cve_id, hostname))
    
    # Compute threat score
    threat_score = 0.0
    if incidents and any(i['severity'] == 'High' for i in incidents):
        threat_score += 0.6
    elif incidents:
        threat_score += 0.3
    if is_running and is_loaded:
        threat_score += 0.4
    elif is_running or is_loaded:
        threat_score += 0.2
    threat_score = min(1.0, threat_score + rng.uniform(-0.1, 0.1))
    
    return {
        "sentinel_incidents": incidents,
        "edr_alerts": alerts,
        "process_evidence": {
            "is_running": is_running,
            "is_loaded": is_loaded,
            "running_processes": [...] if is_running else [],
            "evidence_quality": "high" if is_running and is_loaded else ("medium" if is_running else "low"),
        },
        "threat_score": round(threat_score, 2),
    }
```

**Generation logic:**
```python
def get_network_context(hostname, asset_ctx):
    zone = (asset_ctx.get("network_zone") or "").lower()
    
    # Determine agency from CMDB or hostname parsing
    agency = _extract_agency_from_hostname(hostname)
    
    # Query synthetic_assets for peers (fall back to heuristic if table not populated)
    try:
        from database import get_db
        db = get_db()
        peers = db.execute(
            'SELECT hostname, is_internet_facing, is_dc, data_classification '
            'FROM synthetic_assets '
            'WHERE agency = ? AND network_zone LIKE ? AND in_holdout = 0',
            (agency, f'%{zone}%')
        ).fetchall() if db else []
        db.close()
    except Exception:
        peers = []  # Fall back to heuristic
    
    if not peers:
        # Heuristic fallback (same as current _synthetic_lateral logic)
        zone_baseline = {"protected": 40, "restricted": 25, "dmz": 15, "management": 8}
        peer_count = zone_baseline.get(zone, 5)
        internet_facing_peers = max(1, peer_count // 10)
        dc_peers = max(0, peer_count // 20)
        classifications = {"OFFICIAL": peer_count}
    else:
        peer_count = len(peers)
        internet_facing_peers = sum(1 for p in peers if p[1])
        dc_peers = sum(1 for p in peers if p[2])
        classifications = {}
        for p in peers:
            dc = p[3] or "Category 1"
            classifications[dc] = classifications.get(dc, 0) + 1
    
    # Compute blast radius score
    blast_radius = min(1.0, (peer_count / 100) * 0.5 + (internet_facing_peers / peer_count) * 0.5) if peer_count > 0 else 0.0
    
    # Exposure level
    if blast_radius > 0.5:  exposure_level = "high"
    elif blast_radius > 0.2: exposure_level = "medium"
    else: exposure_level = "low"
    
    return {
        "zone": zone,
        "zone_display": _get_zone_display(zone, hostname),
        "peer_count": peer_count,
        "internet_facing_peers": internet_facing_peers,
        "dc_peers": dc_peers,
        "peer_classifications": classifications,
        "blast_radius_score": round(blast_radius, 2),
        "exposure_level": exposure_level,
    }
```
